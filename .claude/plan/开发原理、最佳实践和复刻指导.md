### 1.2 核心设计原则

#### **单一职责原则 (SRP)**
- 每个模块专注于特定功能
- 清晰的接口定义和职责边界
- 便于测试和维护

#### **开闭原则 (OCP)**
- 通过接口和抽象支持扩展
- 对修改关闭，对扩展开放
- 插件系统的核心设计理念

#### **依赖倒置原则 (DIP)**
- 高层模块不依赖低层模块实现
- 依赖抽象而非具体实现
- 适配器模式的典型应用

#### **接口隔离原则 (ISP)**
- 使用细粒度接口定义
- 客户端不依赖不需要的接口
- 提高系统的灵活性和可维护性

```typescript
    this.config = this.loadConfig();

    const searchPaths = [
      './mcp-config.json',
      './.mcp/config.json',
      process.env.MCP_CONFIG_PATH
    ];
    
    for (const path of searchPaths) {
      if (fs.existsSync(path)) {
        return path;
      }
    }
    
    throw new Error('No configuration file found');
  }

    try {
      const configData = fs.readFileSync(this.configPath, 'utf8');
      return JSON.parse(configData);
    } catch (error) {
      throw new Error(`Failed to load config: ${error.message}`);
    }
  }
  
  getConfig(): MCPConfig {
    return this.config;
  }
  
  updateConfig(updates: Partial<MCPConfig>): void {
    this.config = { ...this.config, ...updates };
    this.saveConfig();
  }
  
  private saveConfig(): void {
    fs.writeFileSync(this.configPath, JSON.stringify(this.config, null, 2));
  }
```

```typescript
export enum CircuitState {
  CLOSED = 'closed',
  OPEN = 'open',
  HALF_OPEN = 'half_open'
}

export interface CircuitBreakerConfig {
  failureThreshold: number;
  resetTimeout: number;
  monitoringPeriod: number;
}

export class CircuitBreaker {
  private state: CircuitState = CircuitState.CLOSED;
  private failures: number = 0;
  private lastFailureTime: number = 0;
  private nextAttemptTime: number = 0;
  
  constructor(private config: CircuitBreakerConfig) {}
  
  async execute<T>(operation: () => Promise<T>): Promise<T> {
    if (this.state === CircuitState.OPEN) {
      if (Date.now() < this.nextAttemptTime) {
        throw new Error('Circuit breaker is open');
      }
      this.state = CircuitState.HALF_OPEN;
    }
    
    try {
      const result = await operation();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }
  
  private onSuccess(): void {
    this.failures = 0;
    this.state = CircuitState.CLOSED;
  }
  
  private onFailure(): void {
    this.failures++;
    this.lastFailureTime = Date.now();
    
    if (this.failures >= this.config.failureThreshold) {
      this.state = CircuitState.OPEN;
      this.nextAttemptTime = Date.now() + this.config.resetTimeout;
    }
  }
  
  getState(): CircuitState {
    return this.state;
  }
}
```

```typescript
// src/core/intelligent-constraints/constraint-manager.ts
export interface Constraint {
  id: string;
  type: 'validation' | 'transformation' | 'generation';
  target: string;
  pattern: string;
  severity: 'error' | 'warning' | 'info';
  message: string;
}

export class ConstraintManager {
  private constraints: Map<string, Constraint> = new Map();
  private templateEngine: ConstraintTemplateEngine;
  
  constructor() {
    this.templateEngine = new ConstraintTemplateEngine();
  }
  
  addConstraint(constraint: Constraint): void {
    this.constraints.set(constraint.id, constraint);
  }
  
  removeConstraint(id: string): void {
    this.constraints.delete(id);
  }
  
  getConstraints(): Constraint[] {
    return Array.from(this.constraints.values());
  }
  
  async applyConstraints(code: string, context: any): Promise<ConstraintResult> {
    const results: ConstraintResult[] = [];
    
    for (const constraint of this.constraints.values()) {
      try {
        const result = await this.applySingleConstraint(constraint, code, context);
        results.push(result);
      } catch (error) {
        console.error(`Error applying constraint ${constraint.id}:`, error);
      }
    }
    
    return {
      success: results.every(r => r.success),
      results,
      modifiedCode: this.generateModifiedCode(code, results)
    };
  }
  
  private async applySingleConstraint(
    constraint: Constraint, 
    code: string, 
    context: any
  ): Promise<ConstraintResult> {
    // 使用模板引擎处理约束
    const processedPattern = await this.templateEngine.render(
      constraint.pattern, 
      context
    );
    
    const regex = new RegExp(processedPattern, 'g');
    const matches = code.match(regex);
    
    return {
      constraintId: constraint.id,
      success: constraint.type === 'validation' ? !matches : true,
      violations: matches || [],
      message: constraint.message,
      severity: constraint.severity
    };
  }
  
  private generateModifiedCode(originalCode: string, results: ConstraintResult[]): string {
    let modifiedCode = originalCode;
    
    for (const result of results) {
      if (!result.success && result.suggestedFix) {
        modifiedCode = modifiedCode.replace(
          new RegExp(result.violations[0], 'g'),
          result.suggestedFix
        );
      }
    }
    
    return modifiedCode;
  }
}
```

```typescript
// src/core/mcp-server/index.ts
export class MCPServer {
  private app: Express;
  private server: http.Server;
  private wsServer?: WebSocket.Server;
  private config: MCPConfig;
  private logger: Logger;
  private configManager: ConfigManager;
  private validationEngine: ValidationEngine;
  private sessionManager: SessionManager;
  
  constructor(configPath?: string) {
    this.configManager = new ConfigManager(configPath);
    this.config = this.configManager.getConfig();
    this.logger = new Logger();
    this.validationEngine = new ValidationEngine(this.config.validation);
    this.sessionManager = new SessionManager();
    
    this.app = express();
    this.setupMiddleware();
    this.setupRoutes();
  }
  
  private setupMiddleware(): void {
    this.app.use(cors());
    this.app.use(express.json({ limit: '10mb' }));
    this.app.use(this.requestLogger);
  }
  
  private setupRoutes(): void {
    // 健康检查
    this.app.get('/health', (req, res) => {
      res.json({
        status: 'healthy',
        timestamp: new Date().toISOString(),
        version: this.config.server.version
      });
    });
    
    // MCP端点
    this.app.post('/mcp', async (req, res) => {
      try {
        const response = await this.handleMCPRequest(req.body);
        res.json(response);
      } catch (error) {
        res.status(500).json({
          jsonrpc: '2.0',
          error: {
            code: -32603,
            message: error.message
          }
        });
      }
    });
  }
  
  private async handleMCPRequest(request: MCPMessage): Promise<MCPMessage> {
    this.logger.info('MCP Request', { method: request.method, id: request.id });
    
    switch (request.method) {
      case 'initialize':
        return this.handleInitialize(request);
      case 'tools/list':
        return this.handleListTools(request);
      case 'tools/call':
        return this.handleCallTool(request);
      default:
        throw new Error(`Unknown method: ${request.method}`);
    }
  }
  
  private handleInitialize(request: MCPMessage): MCPMessage {
    return {
      jsonrpc: '2.0',
      id: request.id,
      result: {
        protocolVersion: '2024-11-05',
        capabilities: {
          tools: { listChanged: true }
        },
        serverInfo: {
          name: this.config.server.name,
          version: this.config.server.version
        }
      }
    };
  }
  
  private handleListTools(request: MCPMessage): MCPMessage {
    const tools = [
      {
        name: 'validate_interface',
        description: 'Validate API interface against OpenAPI specification',
        inputSchema: {
          type: 'object',
          properties: {
            specification: { type: 'string' },
            interfaceData: { type: 'object' },
            options: { type: 'object' }
          },
          required: ['specification', 'interfaceData']
        }
      }
    ];
    
    return {
      jsonrpc: '2.0',
      id: request.id,
      result: { tools }
    };
  }
  
  private async handleCallTool(request: MCPMessage): Promise<MCPMessage> {
    const { name, arguments: args } = request.params;
    
    switch (name) {
      case 'validate_interface':
        const result = await this.validationEngine.validate(args);
        return {
          jsonrpc: '2.0',
          id: request.id,
          result: {
            content: [{ type: 'text', text: JSON.stringify(result, null, 2) }]
          }
        };
      default:
        throw new Error(`Unknown tool: ${name}`);
    }
  }
  
  async start(): Promise<void> {
    const port = process.env.PORT || 3000;
    
    this.server = this.app.listen(port, () => {
      this.logger.info(`MCP Server listening on port ${port}`);
    });
    
    // 启用WebSocket支持
    this.wsServer = new WebSocket.Server({ server: this.server });
    this.setupWebSocketHandlers();
  }
  
  async stop(): Promise<void> {
    if (this.server) {
      this.server.close();
    }
    if (this.wsServer) {
      this.wsServer.close();
    }
  }
  
  private setupWebSocketHandlers(): void {
    if (!this.wsServer) return;
    
    this.wsServer.on('connection', (ws, req) => {
      const sessionId = this.sessionManager.createSession(req);
      
      ws.on('message', async (data) => {
        try {
          const message = JSON.parse(data.toString());
          const response = await this.handleMCPRequest(message);
          ws.send(JSON.stringify(response));
        } catch (error) {
          ws.send(JSON.stringify({
            jsonrpc: '2.0',
            error: { code: -32603, message: error.message }
          }));
        }
      });
      
      ws.on('close', () => {
        this.sessionManager.removeSession(sessionId);
      });
    });
  }
  
  private requestLogger = (req: express.Request, res: express.Response, next: express.NextFunction) => {
    this.logger.info(`${req.method} ${req.path}`, {
      ip: req.ip,
      userAgent: req.get('User-Agent')
    });
    next();
  };
}
```

```typescript
class CacheManager {
  private cache = new Map<string, CacheEntry>();
  private cleanupInterval: NodeJS.Timeout;
  
  constructor(private ttl: number = 300000) {
    this.cleanupInterval = setInterval(() => this.cleanup(), 60000);
  }
  
  set(key: string, value: any, customTtl?: number): void {
    const expiry = Date.now() + (customTtl || this.ttl);
    this.cache.set(key, { value, expiry });
  }
  
  get(key: string): any | null {
    const entry = this.cache.get(key);
    if (!entry) return null;
    
    if (Date.now() > entry.expiry) {
      this.cache.delete(key);
      return null;
    }
    
    return entry.value;
  }
  
  private cleanup(): void {
    const now = Date.now();
    for (const [key, entry] of this.cache.entries()) {
      if (now > entry.expiry) {
        this.cache.delete(key);
      }
    }
  }
}

// 使用缓存装饰器
function cache(ttl: number = 300000) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    const cache = new CacheManager(ttl);
    
    descriptor.value = async function (...args: any[]) {
      const key = `${propertyKey}:${JSON.stringify(args)}`;
      const cached = cache.get(key);
      
      if (cached) return cached;
      
      const result = await originalMethod.apply(this, args);
      cache.set(key, result);
      return result;
    };
  };
}
```

```typescript
  private maxPoolSize: number;

    if (this.pool.length > 0) {
      return this.pool.pop();
    }
    
    if (this.activeConnections < this.maxPoolSize) {
      this.activeConnections++;
      return this.createConnection();
    }
    
    // 等待连接可用
    return new Promise((resolve) => {
      const checkConnection = () => {
        if (this.pool.length > 0) {
          resolve(this.pool.pop());
        } else {
          setTimeout(checkConnection, 100);
        }
      };
      checkConnection();
    });
  }
  
  releaseConnection(connection: any): void {
    if (this.pool.length < this.maxPoolSize) {
      this.pool.push(connection);
    } else {
      connection.close();
      this.activeConnections--;
    }
  }
```

```typescript
// 批量处理器
class BatchProcessor<T, R> {
  private queue: T[] = [];
  private processing = false;
  private batchSize: number;
  private flushInterval: number;
  private processor: (items: T[]) => Promise<R[]>;
  
  constructor(
    processor: (items: T[]) => Promise<R[]>,
    batchSize: number = 100,
    flushInterval: number = 1000
  ) {
    this.processor = processor;
    this.batchSize = batchSize;
    this.flushInterval = flushInterval;
    this.startFlushTimer();
  }
  
  async add(item: T): Promise<void> {
    this.queue.push(item);
    
    if (this.queue.length >= this.batchSize && !this.processing) {
      await this.flush();
    }
  }
  
  private async flush(): Promise<void> {
    if (this.queue.length === 0 || this.processing) return;
    
    this.processing = true;
    const batch = this.queue.splice(0, this.batchSize);
    
    try {
      await this.processor(batch);
    } catch (error) {
      console.error('Batch processing failed:', error);
      // 重新入队失败的项
      this.queue.unshift(...batch);
    } finally {
      this.processing = false;
    }
  }
  
  private startFlushTimer(): void {
    setInterval(() => {
      if (!this.processing) {
        this.flush();
      }
    }, this.flushInterval);
  }
}
```

```typescript
class RateLimiter {
  private requests = new Map<string, number[]>();
  private windowSize: number;
  private maxRequests: number;
  
  constructor(windowSize: number = 60000, maxRequests: number = 100) {
    this.windowSize = windowSize;
    this.maxRequests = maxRequests;
  }
  
  isAllowed(clientId: string): boolean {
    const now = Date.now();
    const windowStart = now - this.windowSize;
    
    if (!this.requests.has(clientId)) {
      this.requests.set(clientId, []);
    }
    
    const clientRequests = this.requests.get(clientId)!;
    const validRequests = clientRequests.filter(time => time > windowStart);
    
    if (validRequests.length >= this.maxRequests) {
      return false;
    }
    
    validRequests.push(now);
    this.requests.set(clientId, validRequests);
    return true;
  }
}

const rateLimiter = new RateLimiter();

const rateLimitMiddleware = (req: express.Request, res: express.Response, next: express.NextFunction) => {
  const clientId = req.ip || req.connection.remoteAddress || 'unknown';
  
  if (!rateLimiter.isAllowed(clientId)) {
    return res.status(429).json({
      error: {
        code: 'RATE_LIMIT_EXCEEDED',
        message: 'Too many requests'
      }
    });
  }
  
  next();
```
