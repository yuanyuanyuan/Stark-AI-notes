```typescript
class PluginLifecycleManager {
    private stateMachine: Map<PluginState, PluginState[]> = new Map([
        [PluginState.LOADED, [PluginState.INITIALIZED, PluginState.ERROR]],
        [PluginState.INITIALIZED, [PluginState.STARTED, PluginState.STOPPED, PluginState.ERROR]],
        [PluginState.STARTED, [PluginState.STOPPED, PluginState.ERROR]],
        [PluginState.STOPPED, [PluginState.STARTED, PluginState.DESTROYED, PluginState.ERROR]],
        [PluginState.ERROR, [PluginState.LOADED, PluginState.DESTROYED]]
    ]);
    
    async transitionState(
        plugin: IPlugin, 
        targetState: PluginState
    ): Promise<void> {
        const currentState = plugin.getState();
        const allowedTransitions = this.stateMachine.get(currentState) || [];
        
        if (!allowedTransitions.includes(targetState)) {
            throw new Error(`Invalid state transition from ${currentState} to ${targetState}`);
        }
        
        await this.executeStateTransition(plugin, currentState, targetState);
    }
    
    private async executeStateTransition(
        plugin: IPlugin, 
        from: PluginState, 
        to: PluginState
    ): Promise<void> {
        switch (to) {
            case PluginState.INITIALIZED:
                await plugin.initialize();
                break;
            case PluginState.STARTED:
                await plugin.start();
                break;
            case PluginState.STOPPED:
                await plugin.stop();
                break;
            case PluginState.DESTROYED:
                await plugin.destroy();
                break;
        }
        
        plugin.setState(to);
    }
}
```

#### 2. **依赖解析系统**
```typescript
class DependencyResolver {
    private dependencyGraph: Map<string, string[]> = new Map();
    
    // 解析插件依赖
    async resolveDependencies(plugins: PluginManifest[]): Promise<string[]> {
        const loadOrder: string[] = [];
        const visited = new Set<string>();
        const visiting = new Set<string>();
        
        for (const plugin of plugins) {
            if (!visited.has(plugin.id)) {
                await this.topologicalSort(plugin.id, plugins, visited, visiting, loadOrder);
            }
        }
        
        return loadOrder;
    }
    
    private async topologicalSort(
        pluginId: string,
        plugins: PluginManifest[],
        visited: Set<string>,
        visiting: Set<string>,
        loadOrder: string[]
    ): Promise<void> {
        if (visiting.has(pluginId)) {
            throw new Error(`Circular dependency detected: ${pluginId}`);
        }
        
        if (visited.has(pluginId)) {
            return;
        }
        
        visiting.add(pluginId);
        
        const plugin = plugins.find(p => p.id === pluginId);
        if (plugin) {
            for (const depId of Object.keys(plugin.dependencies || {})) {
                await this.topologicalSort(depId, plugins, visited, visiting, loadOrder);
            }
        }
        
        visiting.delete(pluginId);
        visited.add(pluginId);
        loadOrder.push(pluginId);
    }
    
    // 验证依赖版本
    validateVersion(dependency: string, required: string, available: string): boolean {
        // 实现语义版本验证
        return this.satisfies(available, required);
    }
}
```

#### 1. **插件事件系统**
```typescript
class PluginEventSystem extends EventEmitter {
    private eventHandlers: Map<string, Function[]> = new Map();
    private middleware: Function[] = [];
    
    // 注册事件处理器
    on(event: string, handler: Function, priority: number = 0): void {
        if (!this.eventHandlers.has(event)) {
            this.eventHandlers.set(event, []);
        }
        
        const handlers = this.eventHandlers.get(event)!;
        handlers.push({ handler, priority });
        handlers.sort((a, b) => b.priority - a.priority);
    }
    
    // 触发事件
    async emit(event: string, ...args: any[]): Promise<any[]> {
        const results: any[] = [];
        
        // 执行中间件
        let processedArgs = args;
        for (const middleware of this.middleware) {
            processedArgs = await middleware(event, ...processedArgs);
        }
        
        // 执行事件处理器
        const handlers = this.eventHandlers.get(event) || [];
        for (const { handler } of handlers) {
            try {
                const result = await handler(...processedArgs);
                results.push(result);
            } catch (error) {
                this.emit('plugin:error', { event, error, handler });
            }
        }
        
        return results;
    }
    
    // 添加中间件
    use(middleware: Function): void {
        this.middleware.push(middleware);
    }
}
```

```typescript
class PluginHookSystem {
    private hooks: Map<string, PluginHook[]> = new Map();
    
    // 注册钩子
    registerHook(hookName: string, handler: Function, options: HookOptions = {}): void {
        if (!this.hooks.has(hookName)) {
            this.hooks.set(hookName, []);
        }
        
        const hook: PluginHook = {
            handler,
            priority: options.priority || 0,
            condition: options.condition,
            pluginId: options.pluginId || 'unknown'
        };
        
        const hooks = this.hooks.get(hookName)!;
        hooks.push(hook);
        hooks.sort((a, b) => b.priority - a.priority);
    }
    
    // 执行钩子
    async executeHooks(hookName: string, context: any): Promise<any> {
        const hooks = this.hooks.get(hookName) || [];
        let result = context;
        
        for (const hook of hooks) {
            if (this.shouldExecuteHook(hook, context)) {
                try {
                    result = await hook.handler(result, context);
                } catch (error) {
                    console.error(`Hook execution failed: ${hookName}`, error);
                }
            }
        }
        
        return result;
    }
    
    private shouldExecuteHook(hook: PluginHook, context: any): boolean {
        if (!hook.condition) return true;
        
        try {
            return new Function('context', `return ${hook.condition}`)(context);
        } catch (error) {
            console.error('Hook condition evaluation failed:', error);
            return false;
        }
    }
}
```

```typescript
class PluginSandbox {
    private sandbox: vm.Context;
    private allowedModules: Set<string>;
    
    constructor(allowedModules: string[] = []) {
        this.allowedModules = new Set(allowedModules);
        this.sandbox = this.createSandbox();
    }
    
    private createSandbox(): vm.Context {
        const sandbox: vm.Context = {
            console: {
                log: console.log,
                error: console.error,
                warn: console.warn,
                info: console.info
            },
            setTimeout,
            clearTimeout,
            setInterval,

            Promise,

        };

        return sandbox;

            if (!this.allowedModules.has(module)) {
                throw new Error(`Module '${module}' is not allowed in plugin sandbox`);
            }
            return require(module);
        };
    }
    
    // 在沙箱中执行代码
    execute(code: string, filename: string): any {
        try {
            const script = new vm.Script(code, {
                filename,
                timeout: 5000,
                sandbox: this.sandbox
            });
            return script.runInContext(this.sandbox);
        } catch (error) {
            throw new Error(`Plugin execution failed: ${error.message}`);
        }
    }
}
```

```typescript
class PluginPermissionManager {
    private permissions: Map<string, Permission[]> = new Map();
    
    // 定义权限
    definePermission(permission: Permission): void {
        const { resource, actions } = permission;
        const key = this.getPermissionKey(resource, actions);
        this.permissions.set(key, permission);
    }
    
    // 检查权限
    checkPermission(pluginId: string, resource: string, action: string): boolean {
        const key = this.getPermissionKey(resource, [action]);
        const permission = this.permissions.get(key);
        
        if (!permission) {
            return false;
        }
        
        // 检查插件是否具有该权限
        return this.hasPluginPermission(pluginId, permission);
    }
    
    // 授予权限
    grantPermission(pluginId: string, resource: string, actions: string[]): void {
        for (const action of actions) {
            const key = this.getPermissionKey(resource, [action]);
            const permission = this.permissions.get(key);
            
            if (permission) {
                permission.allowedPlugins.add(pluginId);
            }
        }
    }
    
    private getPermissionKey(resource: string, actions: string[]): string {
        return `${resource}:${actions.sort().join(',')}`;
    }
    
    private hasPluginPermission(pluginId: string, permission: Permission): boolean {
        return permission.allowedPlugins.has(pluginId) || permission.allowedPlugins.has('*');
    }
}
```

```typescript
class PluginCache {
    private cache: Map<string, CacheEntry> = new Map();
    private cleanupInterval: NodeJS.Timeout;
    
    constructor(ttl: number = 300000) { // 5分钟默认TTL
        this.cleanupInterval = setInterval(() => {
            this.cleanup();
        }, 60000); // 每分钟清理一次
    }
    
    // 设置缓存
    set(key: string, value: any, ttl?: number): void {
        const expiry = Date.now() + (ttl || 300000);
        this.cache.set(key, { value, expiry });
    }
    
    // 获取缓存
    get(key: string): any | null {
        const entry = this.cache.get(key);
        
        if (!entry) {
            return null;
        }
        
        if (Date.now() > entry.expiry) {
            this.cache.delete(key);
            return null;
        }
        
        return entry.value;
    }
    
    // 清理过期缓存
    private cleanup(): void {
        const now = Date.now();
        for (const [key, entry] of this.cache.entries()) {
            if (now > entry.expiry) {
                this.cache.delete(key);
            }
        }
    }
    
    // 清除缓存
    clear(): void {
        this.cache.clear();
    }
    
    // 销毁缓存
    destroy(): void {
        clearInterval(this.cleanupInterval);
        this.cache.clear();
    }
}
```

```typescript
class PluginPool {
    private pools: Map<string, PluginPoolEntry[]> = new Map();
    private maxPoolSize: number = 10;
    
    // 获取插件实例
    async getPlugin(pluginId: string): Promise<IPlugin> {
        if (!this.pools.has(pluginId)) {
            this.pools.set(pluginId, []);
        }
        
        const pool = this.pools.get(pluginId)!;
        
        // 查找可用实例
        const availableInstance = pool.find(entry => entry.available);
        
        if (availableInstance) {
            availableInstance.available = false;
            availableInstance.lastUsed = Date.now();
            return availableInstance.instance;
        }
        
        // 创建新实例
        if (pool.length < this.maxPoolSize) {
            const instance = await this.createPluginInstance(pluginId);
            pool.push({
                instance,
                available: false,
                lastUsed: Date.now()
            });
            return instance;
        }
        
        // 回收最久未使用的实例
        const oldestEntry = pool.reduce((oldest, current) => 
            current.lastUsed < oldest.lastUsed ? current : oldest
        );
        
        oldestEntry.available = false;
        oldestEntry.lastUsed = Date.now();
        return oldestEntry.instance;
    }
    
    // 释放插件实例
    releasePlugin(pluginId: string, instance: IPlugin): void {
        const pool = this.pools.get(pluginId);
        if (pool) {
            const entry = pool.find(e => e.instance === instance);
            if (entry) {
                entry.available = true;
            }
        }
    }
    
    // 清理空闲实例
    cleanup(): void {
        for (const [pluginId, pool] of this.pools.entries()) {
            const now = Date.now();
            const idleThreshold = 300000; // 5分钟
            
            const activeInstances = pool.filter(entry => 
                !entry.available || (now - entry.lastUsed) < idleThreshold
            );
            
            this.pools.set(pluginId, activeInstances);
        }
    }
}
```

```typescript
class PluginExtensionPoints {
    private extensionPoints: Map<string, ExtensionPoint[]> = new Map();
    
    // 注册扩展点
    registerExtensionPoint(name: string, config: ExtensionPointConfig): void {
        const extensionPoint: ExtensionPoint = {
            name,
            config,
            extensions: []
        };
        
        this.extensionPoints.set(name, extensionPoint);
    }
    
    // 注册扩展
    registerExtension(pointName: string, extension: Extension): void {
        const point = this.extensionPoints.get(pointName);
        if (!point) {
            throw new Error(`Extension point '${pointName}' not found`);
        }
        
        // 验证扩展
        this.validateExtension(extension, point.config);
        
        point.extensions.push(extension);
        point.extensions.sort((a, b) => (b.priority || 0) - (a.priority || 0));
    }
    
    // 执行扩展
    async executeExtensions(pointName: string, context: any): Promise<any> {
        const point = this.extensionPoints.get(pointName);
        if (!point) {
            return context;
        }
        
        let result = context;
        for (const extension of point.extensions) {
            if (this.shouldExecuteExtension(extension, context)) {
                try {
                    result = await extension.execute(result, context);
                } catch (error) {
                    console.error(`Extension execution failed: ${pointName}`, error);
                }
            }
        }
        
        return result;
    }
    
    private validateExtension(extension: Extension, config: ExtensionPointConfig): void {
        // 实现扩展验证逻辑
        if (config.requiredMethods) {
            for (const method of config.requiredMethods) {
                if (typeof extension[method] !== 'function') {
                    throw new Error(`Extension missing required method: ${method}`);
                }
            }
        }
    }
    
    private shouldExecuteExtension(extension: Extension, context: any): boolean {
        if (!extension.condition) return true;
        
        try {
            return extension.condition(context);
        } catch (error) {
            console.error('Extension condition evaluation failed:', error);
            return false;
        }
    }
}
```

```typescript
class PluginHotReloader {
    private watchers: Map<string, FSWatcher> = new Map();
    private reloadQueue: Map<string, NodeJS.Timeout> = new Map();
    
    // 监控插件变化
    watchPlugin(pluginId: string, pluginPath: string): void {
        if (this.watchers.has(pluginId)) {
            return;
        }
        
        const watcher = fs.watch(pluginPath, { recursive: true }, (eventType, filename) => {
            if (filename) {
                this.scheduleReload(pluginId);
            }
        });
        
        this.watchers.set(pluginId, watcher);
    }
    
    // 调度重载
    private scheduleReload(pluginId: string): void {
        // 防抖处理
        if (this.reloadQueue.has(pluginId)) {
            clearTimeout(this.reloadQueue.get(pluginId)!);
        }
        
        const timeout = setTimeout(() => {
            this.reloadPlugin(pluginId);
        }, 1000); // 1秒延迟
        
        this.reloadQueue.set(pluginId, timeout);
    }
    
    // 重载插件
    private async reloadPlugin(pluginId: string): Promise<void> {
        try {
            console.log(`Reloading plugin: ${pluginId}`);
            
            // 停止插件
            await this.pluginManager.stopPlugin(pluginId);
            
            // 卸载插件
            await this.pluginManager.unloadPlugin(pluginId);
            
            // 重新发现插件
            await this.pluginManager.discoverPlugins();
            
            // 重新加载插件
            await this.pluginManager.loadPlugin(pluginId);
            
            // 启动插件
            await this.pluginManager.startPlugin(pluginId);
            
            console.log(`Plugin reloaded successfully: ${pluginId}`);
        } catch (error) {
            console.error(`Plugin reload failed: ${pluginId}`, error);
        }
    }
    
    // 停止监控
    unwatchPlugin(pluginId: string): void {
        const watcher = this.watchers.get(pluginId);
        if (watcher) {
            watcher.close();
            this.watchers.delete(pluginId);
        }
        
        const timeout = this.reloadQueue.get(pluginId);
        if (timeout) {
            clearTimeout(timeout);
            this.reloadQueue.delete(pluginId);
        }
    }
}
```

```typescript
                metrics.executionCount++;
                metrics.lastExecutionTime = Date.now();
                metrics.averageExecutionTime = this.calculateAverage(
                    metrics.averageExecutionTime, 
                    metrics.executionCount, 
                    metric.value
                );
                break;
            case 'error':
                metrics.errorCount++;
                this.checkErrorThreshold(pluginId, metrics);
                break;
            case 'memory':
                metrics.memoryUsage = metric.value;
                this.checkMemoryThreshold(pluginId, metrics);
                break;
            case 'cpu':
                metrics.cpuUsage = metric.value;
                this.checkCpuThreshold(pluginId, metrics);
                break;
        }
    }
    
    // 检查阈值
    private checkErrorThreshold(pluginId: string, metrics: PluginMetrics): void {
        const errorRate = metrics.errorCount / metrics.executionCount;
        if (errorRate > 0.1) { // 10%错误率阈值
            this.createAlert(pluginId, 'high_error_rate', `Error rate: ${(errorRate * 100).toFixed(2)}%`);
        }
    }
    
    private checkMemoryThreshold(pluginId: string, metrics: PluginMetrics): void {
        if (metrics.memoryUsage > 100 * 1024 * 1024) { // 100MB阈值
            this.createAlert(pluginId, 'high_memory_usage', `Memory usage: ${metrics.memoryUsage} bytes`);
        }
    }
    
    private checkCpuThreshold(pluginId: string, metrics: PluginMetrics): void {
        if (metrics.cpuUsage > 80) { // 80% CPU使用率阈值
            this.createAlert(pluginId, 'high_cpu_usage', `CPU usage: ${metrics.cpuUsage}%`);
        }
    }
    
    private createAlert(pluginId: string, type: string, message: string): void {
        const alert: PluginAlert = {
            id: `${pluginId}-${type}-${Date.now()}`,
            pluginId,
            type,
            message,
            timestamp: Date.now(),
            resolved: false
        };
        
        this.alerts.push(alert);
        console.warn(`Plugin Alert: ${alert.message}`);
    }
    
    // 获取插件统计
    getPluginStats(pluginId: string): PluginMetrics | null {
        return this.metrics.get(pluginId) || null;
    }
    
    // 获取所有警报
    getAlerts(): PluginAlert[] {
        return this.alerts.filter(alert => !alert.resolved);
    }
}
```
